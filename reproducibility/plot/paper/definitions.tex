\newcommand{\pie}[5][]{
    \begin{scope}[#1]
    \pgfmathsetmacro{\curA}{90}
    \pgfmathsetmacro{\radius}{1}
    \def\Centre{(0,0)}

    \ifx&#2&%
       % #1 is empty
    \else
    \node[pie title, rotate=90] at (180:2.95) {#2}; % #1 is nonempty
    \fi
    \node[pie title] at (90:2) {\textbf{#5}};

    \pgfmathsetmacro{\totalnum}{0}
    \foreach \v/\s/\ox/\oy in {#3} {
        \pgfmathparse{\v+\totalnum}
        \global\let\totalnum=\pgfmathresult
    }

    \newcount\cntr
    \cntr=1 %

    \foreach \v/\s/\ox/\oy in{#3}{
      \pgfmathsetmacro{\deltaA}{\v/\totalnum*360}
      \pgfmathsetmacro{\nextA}{\curA + \deltaA}
      \pgfmathsetmacro{\midA}{(\curA+\nextA)/2}

      \path[slice,\s] \Centre
      -- +(\curA:\radius)
      arc (\curA:\nextA:\radius)
      -- cycle;

      % to determine direction of lines (left/right, up/down
      \pgfmathsetmacro{\ysign}{ifthenelse(mod(\midA,360)<=180,1,-1)}
      \pgfmathsetmacro{\xsign}{ifthenelse(mod(\midA-90,360)<=180,-1,1)}

      \begin{pgfonlayer}{foreground}
        \draw[-,thin] \Centre ++(\midA:\radius/1.1) -- 
        ++(\xsign*0.5*\radius-\ox,\ysign*0.4*\radius-\oy) -- 
        ++(\xsign*\radius,0) 
        node[above,near end,pie values,values of \s] (\the\cntr) {$\v#4$};
      \end{pgfonlayer}

      \node[] (v\the\cntr) at (\the\cntr) {\hphantom{$\v#4$}~};

      \node[right of=v\the\cntr, circle, draw, \s, node distance=.5cm, inner sep=.05cm, minimum size=.1cm] {};
      
      \global\advance\cntr by1
      \global\let\curA\nextA
    }

  \end{scope}
}
